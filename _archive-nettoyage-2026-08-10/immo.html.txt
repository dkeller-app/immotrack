<script>

let properties = JSON.parse(localStorage.getItem('props')||'[]');
let transactions = JSON.parse(localStorage.getItem('tx')||'[]');

function save(){
 localStorage.setItem('props', JSON.stringify(properties));
 localStorage.setItem('tx', JSON.stringify(transactions));
 render();
}

function addProperty(){
 properties.push({
  name: name.value,
  city: city.value,
  rent: Number(rent.value)
 });
 save();
}

function addTx(){
 transactions.push({
  desc: desc.value,
  amount: Number(amount.value)
 });
 save();
}

function render(){

 let income = properties.reduce((a,b)=>a+b.rent,0);
 let expenses = transactions.reduce((a,b)=>a+b.amount,0);
 let cash = income + expenses;

 cashflow.innerText = cash+"€";
 incomeEl = document.getElementById('income');
 incomeEl.innerText = income+"€";
 expensesEl = document.getElementById('expenses');
 expensesEl.innerText = expenses+"€";
 count.innerText = properties.length;

 forecastValue.innerText = (cash*12)+"€";

 // properties
 list.innerHTML = properties.map((p,i)=>
 `<div class='bg-gray-800 p-3 rounded cursor-pointer' onclick='selectProperty(${i})'>
 <h3>${p.name}</h3>
 <p>${p.city}</p>
 <p class='text-green-400'>${p.rent}€</p>
 </div>`).join('');

 // tx
 txlist.innerHTML = transactions.map(t=>
 `<li>${t.desc} : ${t.amount}€</li>`).join('');

 // insights
 insights.innerHTML = "";
 if(expenses < -1000){
 insights.innerHTML += "<li>⚠️ Charges élevées</li>";
 }
 if(income < 2000){
 insights.innerHTML += "<li>📈 Potentiel augmentation loyers</li>";
 }

 renderChart(income, expenses);
}

let selected = null;

function selectProperty(i){
 selected = properties[i];
 detailContent.innerHTML = `
 <h3 class='text-xl font-bold'>${selected.name}</h3>
 <p>${selected.city}</p>
 <p>Loyer: ${selected.rent}€</p>
 <p class='text-green-400'>Cashflow estimé: ${selected.rent - 200}€</p>
 `;
 showPage('detail');
}

let chart;
function renderChart(income, expenses){
 if(chart) chart.destroy();
 chart = new Chart(document.getElementById('chart'),{
  type:'bar',
  data:{
    labels:['Revenus','Dépenses'],
    datasets:[{data:[income,Math.abs(expenses)]}]
  }
 });
}

function showPage(id){
 document.querySelectorAll('.page').forEach(p=>p.classList.add('hidden'));
 document.getElementById(id).classList.remove('hidden');
}

render();

</script>
